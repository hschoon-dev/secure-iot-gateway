```python
# NEW
def add_affected_endpoints_to_xml(xml_data, affected_endpoints) -> str:
    """
    Return the acceptance ruleset XML extended with affected endpoint stubs.

    The result is returned as an in-memory string; no file is written.

    Previously this function wrote to the fixed path
    `config/Acceptance_Ruleset_modified.xml` before returning. Writing to a
    fixed path is a latent race condition: if two requests are ever processed
    concurrently (e.g. after the processing loop is parallelised), the second
    write corrupts the XML being consumed by the first request. Passing the
    XML string directly through the call chain avoids all file-system
    contention and removes the dependency on acceptance_ruleset_path_modified.
    """
    root = ET.fromstring(xml_data)      # defusedxml — see FIX-04

    affected_endpoints_elem = _ET.SubElement(root, "affected_endpoints")

    for endpoint in affected_endpoints:
        endpoint_elem = _ET.SubElement(affected_endpoints_elem, "affected_endpoint")
        name_elem = _ET.SubElement(endpoint_elem, "name")
        name_elem.text = endpoint[0]
        value_elem = _ET.SubElement(endpoint_elem, "value")
        value_elem.text = "0"

    return _ET.tostring(root, encoding="utf-8").decode("utf-8")
```

## FIX-22 · [MEDIUM] `grpc_request_handler.py` creates a new TLS channel per request

**File:** `interface-partition/grpc_request_handler/grpc_request_handler.py`

**Replace the entire file:**

```python
"""
interface-partition/grpc_request_handler/grpc_request_handler.py

Sends VoR requests to the intermediate-vor-partition via mTLS gRPC.

A persistent GrpcRequestSender is module-level so the TLS handshake and
channel setup happen once per process lifetime, not once per request.
The previous implementation opened a new grpc.secure_channel() inside a
'with' block on every call, incurring a full TLS negotiation round-trip
for every submitted request.
"""

from __future__ import annotations

import logging
import threading
from pathlib import Path
from typing import Any, Optional

import grpc

import request_pb2, request_pb2_grpc

from config_loader import GrpcClientConfig

_logger = logging.getLogger(__name__)


class GrpcRequestSender:
    """
    Persistent mTLS gRPC sender for VoR request submissions.

    Reuses the channel and stub across calls. Thread-safe via RLock.
    The channel is created lazily on the first send() call so that import-time
    certificate errors surface at call time rather than at module load.
    """

    def __init__(self, config: GrpcClientConfig) -> None:
        self._config = config
        self._lock = threading.RLock()
        self._channel: Optional[grpc.Channel] = None
        self._stub: Optional[request_pb2_grpc.RequestServiceStub] = None

    def _load_credentials(self) -> grpc.ChannelCredentials:
        certs = self._config.certs
        try:
            client_key  = Path(certs.client_key).read_bytes()
            client_cert = Path(certs.client_cert).read_bytes()
            ca_cert     = Path(certs.ca_cert).read_bytes()
        except Exception as e:
            _logger.error("Error reading TLS files: %s", e)
            raise
        return grpc.ssl_channel_credentials(
            root_certificates=ca_cert,
            private_key=client_key,
            certificate_chain=client_cert,
        )

    def _ensure_connected(self) -> request_pb2_grpc.RequestServiceStub:
        with self._lock:
            if self._stub is not None:
                return self._stub
            target = f"{self._config.target_host}:{self._config.port}"
            creds = self._load_credentials()
            self._channel = grpc.secure_channel(target, creds)
            self._stub = request_pb2_grpc.RequestServiceStub(self._channel)
            _logger.info("gRPC request channel established to %s", target)
            return self._stub

    def _reset(self) -> None:
        with self._lock:
            ch, self._channel, self._stub = self._channel, None, None
        if ch is not None:
            try:
                ch.close()
            except Exception:
                pass

    def send(self, request: dict[str, Any]) -> None:
        try:
            stub = self._ensure_connected()
        except Exception:
            _logger.error("Cannot send request: channel setup failed.")
            return

        req = request_pb2.SubmitRequest(
            issuer_id              = request["issuer_id"],
            issuer_characteristics = request["issuer_characteristics"],
            timestamp              = request["timestamp"],
            description            = request["description"],
            impact                 = request["impact"],
            parameters             = request["parameters"],
            modification           = request["modification"],
            priority               = int(request["prio"]),
            request_id             = request["request_id"],
        )
        try:
            _logger.debug("Sending request via gRPC: %s", request)
            resp = stub.Submit(req, timeout=self._config.timeout_seconds)
            _logger.info("ACK: %s  server_ts: %s", resp.ack, resp.server_timestamp)
        except grpc.RpcError as e:
            _logger.error("RPC failed (%s): %s", e.code().name, e.details())
            self._reset()
        except Exception as e:
            _logger.error("Unexpected gRPC send error: %s", e)
            self._reset()


# Module-level singleton — initialised by interface_setup.py when it calls
# init_grpc_request_sender(config). send_request_as_grpc() is the public API
# kept for backward compatibility with server_methods.py.
_sender: Optional[GrpcRequestSender] = None


def init_grpc_request_sender(config: GrpcClientConfig) -> None:
    """
    Initialise the module-level sender. Call this once from interface_setup
    before any submit_request() calls are made.
    """
    global _sender
    _sender = GrpcRequestSender(config)


def send_request_as_grpc(request: dict[str, Any], config: GrpcClientConfig) -> None:
    """
    Backward-compatible entry point used by server_methods.submit_request().

    On first call (or if the sender was not pre-initialised), a new sender is
    created from config. Subsequent calls reuse the existing channel.
    """
    global _sender
    if _sender is None:
        _sender = GrpcRequestSender(config)
    _sender.send(request)
```

## BUG-02 · CPC — gRPC timeout cascade kills Modbus data flow to all targets

**Partitions:** `cpc-partition` → `interface-partition`, `psmo-partition`  
**Symptom:** No Modbus data received on either target. OPC UA data to `psmo-partition` works; everything to `interface-partition` times out.

**Cause — two interacting bugs:**

**A) `time.sleep(backoff)` held under `RLock` in `GrpcJsonSender._ensure_connected()`**

```python
# grpc_sender.py
def _ensure_connected(self):
    with self._lock:                        # lock acquired
        ...
        if delta < self._backoff_s:
            time.sleep(self._backoff_s - delta)   # sleep INSIDE lock ← bug
        self._connect_locked()
```

When the interface gRPC send fails and the OPC UA thread enters backoff, the `RLock` is held for the full backoff duration (up to 10 s at steady state).

**B) OPC UA and Modbus threads share the same `GrpcJsonSender` instance per target**

`_SENDER_CACHE` in `grpc_sender.py` is a module-level global. Both the OPC UA service thread and the Modbus service thread call `get_sender("192.168.178.112:50051")` and receive the **same object**. When the OPC UA thread holds the lock during backoff, the Modbus thread's send to the same target blocks on `self._lock.acquire()` for the entire backoff duration — tying up a worker thread from the Modbus loop's `asyncio.to_thread` pool.

With a 1 s polling interval and 5 available thread-pool workers on a 1-CPU VM, blocked workers accumulate within a few cycles and the pool exhausts. No Modbus sends complete.

**Fix — `cpc-partition/grpc_sender/grpc_sender.py`:**

Move the backoff sleep outside the `RLock`:

```python
# OLD
def _ensure_connected(self):
    with self._lock:
        if self._stub is not None:
            return self._stub
        now = time.monotonic()
        delta = now - self._last_failure
        if delta < self._backoff_s:
            time.sleep(self._backoff_s - delta)   # held under lock
        self._connect_locked()
        return self._stub

# NEW
def _ensure_connected(self):
    with self._lock:
        if self._stub is not None:
            return self._stub
        remaining = self._backoff_s - (time.monotonic() - self._last_failure)

    # Sleep OUTSIDE the lock — other callers are not serialised.
    if remaining > 0:
        time.sleep(remaining)

    with self._lock:
        if self._stub is not None:   # re-check after sleep
            return self._stub
        self._connect_locked()
        assert self._stub is not None
        return self._stub
```

Additionally, raise `timeout_s` from 2.0 s to at least 5.0 s in `settings.yaml` to accommodate the TLS handshake latency on the first connection to the interface partition.
